// ============================================================================
//  _1006_TruckScrapTransferWorkflowTests.cs
//  Pruebas del workflow TRANSFER_1006 (traspaso de chatarra) con Temporal + time-skipping.
//
//  Ubicación:  test/OneGate.YardFlow.Workflows.Tests/Gate/_1006_TruckScrapTransferWorkflowTests.cs
//  Correr:     dotnet test --filter "_1006_TruckScrapTransferWorkflowTests"
//
//  Foco: cobertura COMPLETA de _1006_ (traspaso de chatarra entre plantas).
//   - Camino feliz + notificación de LLEGADA a Mercancías en Tránsito (RF-406):
//       CA-406-03 (llegada en check-in antes del cierre), CA-406-04 (MeT no responde →
//       MET_SYNC_PENDING y continúa), CA-406-05 (detalle del viaje persistido).
//   - Radiación (device/manual, hard-stop ALARM), validación de imágenes de origen
//       (confirmar / rechazar / timeout), pre y clasificación final, doble pesaje con
//       respaldos manuales, neto vs pedido (aprobado / timeout), y cierre de traspaso
//       (create + close), incluyendo fallo de negocio y fallo técnico.
//   - Todos los caminos terminales y vencimientos de SLA (con time-skipping).
//
//  Mismos patrones que _1001_/_1003_/_1004_/_1005_: nombres de activity sin sufijo
//  "Async", señales en buffer tras iniciar, DTOs opacos con Uninit<T>(), aprobación
//  de delta con reenvío (GateWorkflowBase resetea el flag al entrar al paso).

        public bool Pesable { get; set; } = true;
        public double EntryWeightKg { get; set; } = 30000.0;
        public double ExitWeightKg { get; set; } = 12000.0;   // neto 18000
        public bool RadiationAlarm { get; set; } = false;
        public bool LlegadaThrows { get; set; } = false;       // CA-406-04: MeT no responde
        public bool EntryScaleFails { get; set; } = false;     // báscula entrada no responde
        public bool ExitScaleFails { get; set; } = false;      // báscula salida no responde
        public bool MovimientoCloses { get; set; } = true;     // false → TRANSFER_CLOSE_FAILED
        public bool AceriaThrows { get; set; } = false;        // true  → WORKFLOW_ERROR (create movimiento)

        /// <summary>Orden de llamadas relevantes (patch:{estado}, llegada, close) para aserciones.</summary>

    }

    private static T Uninit<T>() => (T)RuntimeHelpers.GetUninitializedObject(typeof(T));
    private static void Set(object obj, string prop, object? val) =>
        obj.GetType().GetProperty(prop)!.SetValue(obj, val);

    private static GradingResultDto Grading(string material = "FE-001", string ticket = "TCK-001")
    {
        var g = Uninit<GradingResultDto>();
        Set(g, "PrimaryMaterialCode", material);
        Set(g, "TicketId", ticket);
        try { Set(g, "TripId", "trip-test"); } catch { /* opcional */ }
        return g;
    }

    // ── Runner ──────────────────────────────────────────────────────────────────
    private static async Task<WorkflowResult> RunAsync(
        ScenarioConfig cfg,
        TruckScrapTransferInput input,
        Func<WorkflowHandle<_1006_TruckScrapTransferWorkflow>, Task>? sendSignals = null)
    {
        await using var env = await WorkflowEnvironment.StartTimeSkippingAsync();

        var mocks = new MockActivities(cfg);
        using var worker = new TemporalWorker(
            env.Client,
            new TemporalWorkerOptions(TaskQueue)
                .AddAllActivities(mocks)
                .AddWorkflow<_1006_TruckScrapTransferWorkflow>());

        return await worker.ExecuteAsync(async () =>
        {
            var handle = await env.Client.StartWorkflowAsync(
                (_1006_TruckScrapTransferWorkflow wf) => wf.ExecuteAsync(input),
                new WorkflowOptions($"transfer-1006-{Guid.NewGuid():N}", TaskQueue));

            if (sendSignals is not null)
                await sendSignals(handle);

            return await handle.GetResultAsync();
        });
    }

        PlantId = "CELAYA",
        PlateOrTag = plate,

        ReferenciaTraspasoKg = refKg,
        HumanLoopSlaMinutes = sla,
        WeightDeltaThresholdPercent = 2.0,
    };

    // ── Señales (todas en buffer) ────────────────────────────────────────────────
    private static Task Access(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
        => h.SignalAsync(wf => wf.AccessConfirmedAsync(new AccessConfirmationSignalDto { RealizadoPor = "guardia1" }));
    /// <summary>
    /// Llegada a la báscula: el guardia da el pase y se selecciona la báscula.
    ///
    /// <para>El pase va DENTRO del helper porque en la realidad precede siempre a la selección: sin
    /// el verde el camión no se mueve de la caseta. Desde RF-210 el workflow lo exige (RN-222), así
    /// que separarlos obligaría a repetir la señal en los cinco casos que usan este helper.</para>
    /// </summary>
    private static async Task EntryScale(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
    {
        await h.SignalAsync(wf => wf.PaseAutorizadoAsync());
        await h.SignalAsync(wf => wf.ScaleSelectedForEntryAsync("BASCULA-ENTRADA-01"));
    }
    private static Task CaptureRadiation(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
        => h.SignalAsync(wf => wf.CaptureRadiationResultAsync("RAD-ENTRADA-01"));
    private static Task ImagesConfirmed(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
        => h.SignalAsync(wf => wf.TransferImagesConfirmedAsync());
    private static Task PreGrading(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
        => h.SignalAsync(wf => wf.PreGradingResultReceivedAsync(Grading()));
    private static Task Unloading(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
        => h.SignalAsync(wf => wf.UnloadingCompletedAsync());
    private static Task FinalGrading(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
        => h.SignalAsync(wf => wf.FinalGradingResultReceivedAsync(Grading()));
    private static Task ExitScale(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
        => h.SignalAsync(wf => wf.ScaleSelectedForExitAsync("BASCULA-SALIDA-01"));

    // Secuencia feliz completa (radiación automática + imágenes OK).
    private static async Task DriveHappy(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
    {
        await Access(h);
        await EntryScale(h);
        await CaptureRadiation(h);
        await ImagesConfirmed(h);
        await PreGrading(h);
        await Unloading(h);
        await FinalGrading(h);
        await ExitScale(h);
    }

    [Fact] // TC01 — happy path: completa y notifica llegada a Mercancías en Tránsito
    public async Task HappyPath_NotificaLlegada_Completes()
    {
        var cfg = new ScenarioConfig();
        var res = await RunAsync(cfg, Input(), DriveHappy);

        Assert.True(res.Success);
        Assert.Equal("Completed", res.Data?["FinalState"]?.ToString());

        var calls = cfg.Calls.ToArray();
        // CA-406-03: la llegada a destino se notificó...
        Assert.Contains("llegada", calls);
        // ...y ocurrió ANTES del cierre patrimonial del movimiento (check-in, no salida).
        Assert.True(Array.IndexOf(calls, "llegada") < Array.IndexOf(calls, "close"),
            "La notificación de llegada debe ocurrir en el check-in, antes del cierre del movimiento.");
    }

    [Fact] // TC02 — CA-406-05: el detalle del viaje queda persistido tras el lookup
    public async Task DetalleDeViaje_SePersiste_TrasLookup()
    {
        var cfg = new ScenarioConfig();
        var res = await RunAsync(cfg, Input(plate: "TRAS-002"), DriveHappy);

        Assert.True(res.Success);
        Assert.Contains("patch:TRANSFER_TRIP_DETAIL", cfg.Calls.ToArray());
    }

    [Fact] // TC03 — CA-406-04: Mercancías en Tránsito no responde → MET_SYNC_PENDING, el traspaso continúa
    public async Task LlegadaFalla_MarcaPendienteYContinua()
    {
        var cfg = new ScenarioConfig { LlegadaThrows = true };
        var res = await RunAsync(cfg, Input(plate: "TRAS-003"), DriveHappy);

        Assert.True(res.Success);   // NO aborta
        Assert.Equal("Completed", res.Data?["FinalState"]?.ToString());

        var calls = cfg.Calls.ToArray();
        Assert.Contains("patch:MET_SYNC_PENDING", calls);
        // El traspaso siguió y cerró el movimiento pese al fallo de la notificación.
        Assert.Contains("close", calls);
    }

    [Fact] // TC06 — no regresión: destino no pesable → almacén directo (sin notificar llegada)
    public async Task NonWeighable_Warehouse()
    {
        // La unidad no pesable AHORA PASA POR EL GUARDIA: RF-208 determina el destino "una vez
        // completado el proceso de validación", no antes. Sin Access() agota el timeout de 4h.
        var cfg = new ScenarioConfig { Pesable = false };
        var res = await RunAsync(cfg, Input(plate: "TRAS-006"), Access);
        Assert.True(res.Success);
        Assert.Equal("ALMACEN_DIRECTO", res.Data?["result"]?.ToString());
        // Y se registra A QUÉ almacén va (CA-208-04).
        Assert.False(string.IsNullOrWhiteSpace(res.Data?["assignedZone"]?.ToString()));
        Assert.DoesNotContain("llegada", cfg.Calls.ToArray());
    }

    // ── Señales adicionales para los casos ampliados ─────────────────────────────
    private static Task ImagesRejected(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
        => h.SignalAsync(wf => wf.TransferImagesRejectedAsync("material no coincide con imágenes de origen"));
    private static Task Approve(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h)
        => h.SignalAsync(wf => wf.AdjustmentApprovedAsync());

    // Conduce hasta un punto y deja que el resto expire (para timeouts).
    private static async Task Until(WorkflowHandle<_1006_TruckScrapTransferWorkflow> h, string stop)
    {
        await Access(h);
        if (stop == "afterAccess") return;
        await EntryScale(h);
        if (stop == "afterEntryScale") return;
        await CaptureRadiation(h);
        if (stop == "afterRadiation") return;
        await ImagesConfirmed(h);
        if (stop == "afterImages") return;
        await PreGrading(h);
        if (stop == "afterPreGrading") return;
        await Unloading(h);
        if (stop == "afterUnloading") return;
        await FinalGrading(h);
        if (stop == "afterFinalGrading") return;
        await ExitScale(h);
    }

    {
        // neto = 30000 - 5000 = 25000 vs ref 18000 → fuera de ±2%
        var res = await RunAsync(new ScenarioConfig { ExitWeightKg = 5000.0 },
            Input(plate: "TRAS-021", sla: 100000), async h =>
        {
            await DriveHappy(h);
            for (var i = 0; i < 40; i++)
            {
                try { await h.SignalAsync(wf => wf.AdjustmentApprovedAsync()); }
                catch { break; }
                var desc = await h.DescribeAsync();
                if (desc.Status.ToString() != "Running") break;
                await Task.Delay(20);
            }
        });
        Assert.True(res.Success, $"ErrorCode={res.ErrorCode} | Msg={res.ErrorMessage}");
    }
